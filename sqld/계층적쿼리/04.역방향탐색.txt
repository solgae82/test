=================== 역방향 탐색

SELECT
    MENU_ID,
    MENU_NAME as menu_nm,
    PARENT_MENU_ID as p_menu_id,
    SYS_CONNECT_BY_PATH(menu_name, '>') as path,
    level as lv
FROM TBL_SHOP_MENU
START WITH MENU_ID = 103000	-- 경제/경영 ~ 도서 까지 역방향 탐색
CONNECT BY MENU_ID = PRIOR PARENT_MENU_ID;
