=================== 다중 탐색

SELECT
    MENU_ID,
    MENU_NAME as menu_nm,
    PARENT_MENU_ID as p_menu_id,
    SYS_CONNECT_BY_PATH(menu_name, '>') as path,
    level as lv,
    CONNECT_BY_ROOT MENU_NAME AS ROOT_NM
FROM TBL_SHOP_MENU
START WITH MENU_ID IN (101000, 102000) -- 국내도서, 외국도서 별로 탐색
CONNECT BY PRIOR MENU_ID = PARENT_MENU_ID
ORDER SIBLINGS BY MENU_NAME;
