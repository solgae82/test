regexp_instr

	SELECT REGEXP_INSTR(
	    'AB12CD345EF',  -- 1. 원본 문자열
	    '([0-9]+)',     -- 2. 정규표현식
	    1,              -- 3. 첫 문자부터 검색
	    2,              -- 4. 두 번째 숫자 묶음
	    1,              -- 5. 매칭 끝 다음 위치
	    'c',            -- 6. 대소문자 구별
	    1               -- 7. 첫 번째 괄호 그룹
	) AS RESULT
	FROM DUAL;
	
	옵션	기능
	i	대소문자 구별 안 함
	c	대소문자 구별
	m	^, $를 각 줄의 시작·끝에도 적용
	n	.이 줄바꿈 문자까지 매칭
	x	패턴 안의 공백 문자 무시
