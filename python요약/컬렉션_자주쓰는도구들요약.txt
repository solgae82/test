-- 딕셔너리(Dictionary)에 자주 쓰는 CRUD,메서드 및 키워드들 (14개)
d = {"boy": "소년"} # 예제

d['boy']        # 가져오기
d.get("boy")    # 가져오기

d["girl"] = "소녀"  # 추가(키가 있으면 업데이트)
d.setdefault(k,v)   # 추가(키가 있으면 추가 하지 않음) 

d["boy"] = "수정"   # 수정
d.update(boy='소년')  # 키값이 존재하면 수정, 없으면 추가,{'girl':'소녀'} 형태 가능

del d["boy"]
d.pop("boy")    # ← 값이 아니라 key
d.popitem()	# 마지막 뽑기
d.clear()

d.copy()

d.keys()
d.values()
d.items()
