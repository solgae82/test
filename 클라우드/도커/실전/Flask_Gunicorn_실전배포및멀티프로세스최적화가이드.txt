[Docker] Flask + Gunicorn 실전 배포 및 멀티 프로세스 최적화 가이드

📁 프로젝트 구조 및 소스 코드 표준화
	flask-app/
	├── Dockerfile          # 애플리케이션 이미지 빌드 설계도
	├── docker-compose.yml  # 로컬 개발 및 컨테이너 가동 설정
	├── requirements.txt    # 파이썬 의존성 패키지 리스트
	└── server.py           # Flask 메인 애플리케이션 소스

	1) server.py

		from flask import Flask

		app = Flask(__name__)

		@app.route('/')
		def hello():
		    return "Hello, Docker and Flask!"

		if __name__ == '__main__':
		    # 💡 [중요] docker compose 로컬 볼륨 마운트 개발 환경 전용 코드
		    # 실제 도커 운영(Production) 배포 시에는 아래 gunicorn 구문이 이 코드를 대체하여 실행합니다.
		    app.run(host='0.0.0.0', port=5000)

	2) Dockerfile (캐시 최적화 반영)

		# 파이썬 3.12 슬림 버전으로 시작
		FROM python:3.12-slim
		
		# 컨테이너 내부 작업 폴더 설정
		WORKDIR /app

		# 패키지 설치를 먼저 하여 소스코드가 바뀌어도 pip 설치 과정을 캐싱 처리 (빌드 속도 최적화)
		COPY requirements.txt .
		RUN pip install --no-cache-dir -r requirements.txt

		# 나머지 내 플라스크 소스코드 전부 복사
		COPY . .

		# Gunicorn 웹 서버를 전면에 세워 플라스크 앱 가동
		# server.py 를 실행하는데, app변수가 gunicorn이 다룰 변수명임을 알려줌.
		CMD ["gunicorn", "--bind", "0.0.0.0:5000", "server:app"]


	3) docker-compose.yml
		services:
		  flask-web:
		    build: .                          # 현재 폴더(.)의 Dockerfile을 읽어 이미지를 직접 빌드
		    image: solgae/my-flask-app:1.0    # 빌드 완료 후 도커 엔진에 등록될 이미지 이름:버전
		    container_name: flask_app_dev
		    ports:
		      - "5000:5000"                   # 호스트 포트 5000 : 컨테이너 포트 5000
		    volumes:
		      - .:/app                        # 💡 [개발용 꿀팁] 내 PC 소스 코드 수정본이 컨테이너 내부로 즉시 동기화
		    environment:
		      - FLASK_ENV=development         # 💡 개발 모드 활성화 (디버깅용)


1. [로컬 환경] PC에서 이미지 빌드 및 업로드

	cd D:\flask-app

	# 1. 로컬 환경에서 컨테이너 실행 및 빌드
	docker compose up -d

	# 2. 빌드된 이미지 확인 및 Docker Hub 업로드
	docker login
	docker push solgae/my-flask-app:1.0

2. [운영 환경] 리눅스 서버에서 컨테이너 가동 (오버라이딩)
	운영 서버에서는 이미 빌드된 이미지를 받아와서 컨테이너 설정만 덮어씌워 띄웁니다. 
	따라서 운영 서버의 컴포즈 파일에는 build: . 구문이 들어가지 않는 것이 정석입니다.
	
	예제)
	cd /home/solgae
	vi docker-compose.yml

		services:
		  flask-web:
		    image: solgae/my-flask-app:1.0    # 💡 이미 완성된 이미지를 땡겨옴
		    container_name: flask_app
		    ports:
		      - "5000:5000"
		    environment:
		      - FLASK_ENV=production          # 💡 운영 환경 변수로 덮어쓰기

	# ⚠️ 주의: 운영 환경 컴포즈에는 build 구문이 없으므로 --build 옵션을 빼고 실행합니다.
	docker compose up -d

	---------------------
	명령어 최종 요약 가이드
	
	구버전 명령어 하이픈(-) 규격인 docker-compose는 2023년에 종료되었으므로, 
	반드시 공백 분리형 최신 규격인 docker compose를 사용합니다.
	
	컴포즈 가동 이후에는 컨테이너 삭제(down) 행위 없이 가동 범위만 안전하게 제어합니다.
	
		docker compose stop     # 인프라 전체 안전 정지
		docker compose start    # 인프라 전체 즉시 재개
		docker compose restart  # 인프라 전체 재시작
